<script>
const records=%%RECORDS%%;
const regions=%%REGIONS%%;
const chart=document.getElementById('rent-chart');
const region=document.getElementById('region-filter'), change=document.getElementById('change-filter'), state=document.getElementById('state-filter');
const errors=document.getElementById('error-bars'), labels=document.getElementById('state-labels');
function chooseRows(){
 let selected=records;
 if(change.value!=='all'){
  const ranked=[...records].sort((a,b)=>change.value==='increase'?b.change_pp-a.change_pp:a.change_pp-b.change_pp);
  const ids=new Set(ranked.slice(0,5).map(r=>r.fips));selected=selected.filter(r=>ids.has(r.fips));
 }
 if(region.value!=='all')selected=selected.filter(r=>r.region===region.value);
 if(state.value!=='all')selected=selected.filter(r=>r.fips===state.value);
 return selected;
}
async function render(){
 const selected=chooseRows(), groups=regions.map(g=>selected.filter(r=>r.region===g));
 await Plotly.restyle(chart,{
  x:groups.map(g=>g.map(r=>r.share_2019)),y:groups.map(g=>g.map(r=>r.share_2024)),
  text:groups.map(g=>g.map(r=>r.abbr)),
  customdata:groups.map(g=>g.map(r=>[r.state,r.change_pp,r.change_low90,r.change_high90,r.eligible_2019,r.eligible_2024])),
  'error_x.array':groups.map(g=>g.map(r=>r.moe90_2019)),
  'error_y.array':groups.map(g=>g.map(r=>r.moe90_2024)),
  'error_x.visible':errors.checked,'error_y.visible':errors.checked,
  mode:labels.checked?'markers+text':'markers'
 });
 const view=change.value==='all'?'All changes':change.value==='increase'?'5 largest national estimated increases':'5 largest national estimated decreases';
 document.getElementById('chart-status').textContent=selected.length+' states | '+(region.value==='all'?'All regions':region.value)+' | '+view+(selected.length?'':' | No states match; reset or change filters.');
}
region.addEventListener('change',()=>{state.value='all';return render();});
change.addEventListener('change',()=>{state.value='all';return render();});
state.addEventListener('change',()=>{region.value='all';change.value='all';return render();});
errors.addEventListener('change',render);labels.addEventListener('change',render);
document.getElementById('reset-chart').addEventListener('click',()=>{
 region.value='all';change.value='all';state.value='all';errors.checked=false;labels.checked=false;return render();
});
</script></body></html>
