<script>
const records=%%RECORDS%%;
const chart=document.getElementById('rent-chart');
const metric=document.getElementById('metric-view'),region=document.getElementById('region-filter');
const count=document.getElementById('count-filter'),state=document.getElementById('state-filter'),errors=document.getElementById('error-bars');
const colors={Northeast:'#0072B2',Midwest:'#009E73',South:'#D55E00',West:'#7B3294'};
const symbols={Northeast:'circle',Midwest:'diamond',South:'square',West:'triangle-up'};
function chooseRows(){
 let selected=records;
 if(state.value!=='all')selected=selected.filter(r=>r.fips===state.value);
 else if(region.value!=='all')selected=selected.filter(r=>r.region===region.value);
 const key=metric.value==='change'?'change_pp':'share_2024';
 selected=[...selected].sort((a,b)=>b[key]-a[key]||a.state.localeCompare(b.state));
 const total=selected.length;
 if(state.value==='all'&&count.value!=='all')selected=selected.slice(0,Number(count.value));
 return {selected,total};
}
async function render(){
 const {selected,total}=chooseRows(),isChange=metric.value==='change',n=selected.length;
 await Plotly.update(chart,{
  x:[selected.map(r=>isChange?r.change_pp:r.share_2024)],y:[selected.map((_,i)=>i)],
  customdata:[selected.map(r=>[r.state,r.region,r.share_2019,r.share_2024,r.change_pp,r.change_low90,r.change_high90,r.moe90_2024,r.eligible_2019,r.eligible_2024])],
  'marker.color':[selected.map(r=>colors[r.region])],'marker.symbol':[selected.map(r=>symbols[r.region])],
  'error_x.array':[selected.map(r=>isChange?r.change_moe90_pp:r.moe90_2024)],'error_x.visible':errors.checked
 },{
  'xaxis.title.text':isChange?'Change, 2019-2024<br>(percentage points)':'2024 rent-burden share (%)',
  'xaxis.range':isChange?[-12,12]:[30,66],'xaxis.ticksuffix':isChange?'':'%',
  'xaxis.dtick':isChange?4:5,
  'yaxis.tickvals':selected.map((_,i)=>i),'yaxis.ticktext':selected.map(r=>r.state),
  'yaxis.range':[n-.5,-.5],height:Math.max(390,n*28+240),'shapes[0].visible':isChange
 });
 document.getElementById('chart-status').textContent=n+' of '+total+' matching states | Ranked by '+(isChange?'estimated increase':'2024 burden share')+' | '+(region.value==='all'?'All regions':region.value);
 document.getElementById('interval-note').textContent=isChange?
 'Points show estimated changes in percentage points; horizontal lines are approximate 90% change intervals. The dashed vertical line marks no change. Hover for both years and denominators. These intervals do not establish a definitive ranking.':
 'Points show the 2024 burden share; horizontal lines are approximate 90% intervals for that annual share. Hover still shows the 2019 share and change. The highest burden share in 2024 need not be the largest increase.';
}
metric.addEventListener('change',render);
region.addEventListener('change',()=>{state.value='all';return render();});
count.addEventListener('change',()=>{state.value='all';return render();});
state.addEventListener('change',()=>{region.value='all';return render();});
errors.addEventListener('change',render);
document.getElementById('reset-chart').addEventListener('click',()=>{
 metric.value='change';region.value='all';count.value='10';state.value='all';errors.checked=true;return render();
});
</script></body></html>
